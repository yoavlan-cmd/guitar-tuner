const a={title:nm.a?`${nm.t} - ${nm.a}`:nm.t,artist:nm.a||'',dur:r.dur,bpm:r.bpm,offset:+r.offset.toFixed(1),tuning,segs:r.segs,play:{tun:tuning.id,capo:0},feat:r.feat,...rhythmOf(r),v:2,at:Date.now(),from:'guitars',mix:id}; // no capo is guessed (it can't be heard): the guide takes it from you or the song's chord sheet (gdFit) // kept like a song-video analysis (the last 8 are kept), named after the split const key='mix_'+id; // built again: a capo you set from the video stays try{const old=loadSong(key);if(old&&old.play&&old.play.capoBy==='you'){a.play.capo=old.play.capo;a.play.capoBy='you'}}catch(e){} // and the song's video, when there is one (the capo and the chord shapes in the guitarist's hand) if(it.drive){it.sessStep='video';try{dsStatusUI()}catch(e){}try{await vlLook(it,a)}finally{it.sessStep=''}} localStorage.setItem(KP+'song.'+key,JSON.stringify(compact(a))); const idx=store.get('songs',[]).filter(k=>k!==key);idx.unshift(key);idx.slice(8).forEach(k=>{try{localStorage.removeItem(KP+'song.'+k)}catch(e){}});store.set('songs',idx.slice(0,8)); it.sess='done';it.anaKey=key; try{bkNotify(`“${it.name}”: the session is ready`,'Its chords from the guitars, the Easy and Full guide and the jam track are on its page.')}catch(e){} toast(`“${esc(it.name)}”: the session is ready. `,'ok'); const o=$('asOpen');if(o)o.onclick=()=>{const t=$('toast');if(t)t.hidden=true;asOpenSong(id)}} catch(e){it.sess='failed';try{console.error('auto session',e&&e.message||e)}catch(_){}} await mxIdxSave();try{dsStatusUI()}catch(e){}try{renderLearn()}catch(e){}if(curSec==='session'){try{renderSession()}catch(e){}}} function asOpenSong(id){const g=learnSongs().find(x=>x.r.mix.some(m=>m.mix===id));if(g)openSession(g);else goSec('learn')} // build one by hand (the song's page offers it for a split with no guide, or when the build failed) function asStart(id){const it=mxIdx.find(x=>x.id===id);if(!it||it.state!=='ready')return;if(!AS_ON.includes(it.sess)){it.sess='queued';mxIdxSave();try{dsStatusUI()}catch(e){}}asBuild(id)} // after a restart: finish what was being built; a song added from Drive before sessions built themselves gets its session now mxReady.then(()=>{mxIdx.filter(x=>x.state==='ready'&&(AS_ON.includes(x.sess)||(x.drive&&!x.sess))).forEach(x=>{x.sess='queued';asBuild(x.id)});mxIdxSave()}).catch(()=>{}); window.__as={build:asBuild,start:asStart,list:()=>mkList()};// test hooks
